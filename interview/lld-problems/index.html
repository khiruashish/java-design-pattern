    Low-Level Design (LLD) Problems¶

    Machine-coding and object-oriented design rounds (common at Amazon, Microsoft, Flipkart, Uber, Swiggy, and in senior rounds at service companies) ask you to design and code a small system in 45–90 minutes. Below are 12 classic problems, each with requirements, the patterns that fit, a class diagram, complete runnable code, real output, and the follow-up questions interviewers usually ask.

    A repeatable approach for any LLD round

    1. Clarify requirements (5 min): actors, use cases, what's out of scope. Write them down.
    2. Identify entities and their responsibilities (5 min): nouns become classes, verbs become methods.
    3. Find what varies and isolate it behind an interface (pricing, allocation, notifications): that's where patterns appear.
    4. Code the core flow first, then edge cases, then extensions. Keep it compiling and runnable.
    5. Talk about concurrency, testing and extensibility at the end.

    1. Parking Lot¶

    Requirements: multiple floors; spot sizes small/medium/large; bikes, cars and trucks (a vehicle fits its size or larger); issue a ticket on entry; charge on exit (first 30 minutes free, then per started hour by vehicle type); show free spots; reject reused tickets.     package com.handbook.lld.parkinglot;

import java.time.Duration;
import java.time.Instant;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Optional;

/**
 * Parking lot LLD. Patterns: Strategy (spot allocation, pricing), Factory-style creation of spots
 * per floor, and a small state model per spot (free/occupied).
 */
public final class ParkingLot {

    public enum VehicleType { BIKE, CAR, TRUCK }

    public enum SpotSize {
        SMALL, MEDIUM, LARGE;

        boolean fits(VehicleType v) {
            return switch (v) {
                case BIKE -> true;
                case CAR -> this != SMALL;
                case TRUCK -> this == LARGE;
            };
        }
    }

    public record Vehicle(String plate, VehicleType type) {
    }

    public static final class Spot {
        final String id;
        final int floor;
        final SpotSize size;
        Vehicle parked;

        Spot(String id, int floor, SpotSize size) {
            this.id = id;
            this.floor = floor;
            this.size = size;
        }

        boolean isFree() {
            return parked == null;
        }
    }

    public record Ticket(String id, Vehicle vehicle, String spotId, Instant entry) {
    }

    public record Receipt(String ticketId, String plate, Duration parked, long feePaise) {
    }

    /** Strategy: which free spot to give a vehicle. */
    public interface AllocationStrategy {
        Optional<Spot> choose(List<Spot> spots, Vehicle vehicle);
    }

    /** Strategy: how much to charge. */
    public interface PricingStrategy {
        long feePaise(VehicleType type, Duration parked);
    }

    /** Nearest floor first, and the smallest spot that fits (keeps large spots for trucks). */
    public static AllocationStrategy nearestSmallestFit() {
        return (spots, v) -> spots.stream()
                .filter(Spot::isFree)
                .filter(s -> s.size.fits(v.type()))
                .min((a, b) -> a.floor != b.floor ? Integer.compare(a.floor, b.floor) : a.size.compareTo(b.size));
    }

    /** First 30 minutes free, then a per-hour rate by vehicle type, rounded up to whole hours. */
    public static PricingStrategy hourly(Map<VehicleType, Long> ratePerHourPaise) {
        return (type, parked) -> {
            if (parked.compareTo(Duration.ofMinutes(30)) <= 0) {
                return 0;
            }
            long hours = (parked.toMinutes() + 59) / 60;
            return hours * ratePerHourPaise.get(type);
        };
    }

    private final List<Spot> spots = new ArrayList<>();
    private final Map<String, Ticket> active = new HashMap<>();
    private final Map<String, Spot> spotById = new HashMap<>();
    private final AllocationStrategy allocation;
    private final PricingStrategy pricing;
    private int nextTicket = 1;

    public ParkingLot(AllocationStrategy allocation, PricingStrategy pricing) {
        this.allocation = allocation;
        this.pricing = pricing;
    }

    public ParkingLot addFloor(int floor, int small, int medium, int large) {
        add(floor, SpotSize.SMALL, small);
        add(floor, SpotSize.MEDIUM, medium);
        add(floor, SpotSize.LARGE, large);
        return this;
    }

    private void add(int floor, SpotSize size, int count) {
        for (int i = 1; i <= count; i++) {
            Spot s = new Spot("F" + floor + "-" + size.name().charAt(0) + i, floor, size);
            spots.add(s);
            spotById.put(s.id, s);
        }
    }

    public synchronized Optional<Ticket> park(Vehicle vehicle, Instant at) {
        return allocation.choose(spots, vehicle).map(spot -> {
            spot.parked = vehicle;
            Ticket t = new Ticket("T-" + nextTicket++, vehicle, spot.id, at);
            active.put(t.id(), t);
            return t;
        });
    }

    public synchronized Receipt exit(String ticketId, Instant at) {
        Ticket t = active.remove(ticketId);
        if (t == null) {
            throw new IllegalArgumentException("unknown or already used ticket " + ticketId);
        }
        spotById.get(t.spotId()).parked = null;
        Duration parked = Duration.between(t.entry(), at);
        return new Receipt(t.id(), t.vehicle().plate(), parked, pricing.feePaise(t.vehicle().type(), parked));
    }

    public synchronized long freeSpots(VehicleType type) {
        return spots.stream().filter(Spot::isFree).filter(s -> s.size.fits(type)).count();
    }
}

    package com.handbook.lld.elevator;

import java.util.ArrayList;
import java.util.List;
import java.util.TreeSet;

/**
 * Elevator LLD. Patterns: State (each car's direction), Strategy (which car serves a hall call),
 * and the SCAN ("elevator") algorithm: keep moving in one direction while there are stops ahead.
 */
public final class ElevatorSystem {

    public enum Direction { UP, DOWN, IDLE }

    public static final class Car {
        final String id;
        int floor;
        Direction direction = Direction.IDLE;
        final TreeSet<Integer> stops = new TreeSet<>();

        Car(String id, int floor) {
            this.id = id;
            this.floor = floor;
        }

        /** One tick: move one floor toward the next stop in the current direction (SCAN). */
        String step() {
            if (stops.isEmpty()) {
                direction = Direction.IDLE;
                return id + "@" + floor + " idle";
            }
            if (direction == Direction.IDLE) {
                direction = stops.first() >= floor ? Direction.UP : Direction.DOWN;
            }
            if (direction == Direction.UP && stops.ceiling(floor) == null) {
                direction = Direction.DOWN; // nothing above: reverse
            } else if (direction == Direction.DOWN && stops.floor(floor) == null) {
                direction = Direction.UP;
            }
            if (stops.remove(floor)) {
                return id + "@" + floor + " doors open";
            }
            floor += direction == Direction.UP ? 1 : -1;
            if (stops.remove(floor)) {
                return id + "@" + floor + " doors open";
            }
            return id + "@" + floor + " moving " + direction;
        }
    }

    /** Strategy for assigning a hall call to a car. */
    public interface Dispatcher {
        Car choose(List<Car> cars, int floor, Direction wanted);
    }

    /** Prefer an idle car or one already heading toward the floor in the same direction; then nearest. */
    public static Dispatcher nearestSuitable() {
        return (cars, floor, wanted) -> cars.stream()
                .min((a, b) -> Integer.compare(cost(a, floor, wanted), cost(b, floor, wanted)))
                .orElseThrow();
    }

    private static int cost(Car c, int floor, Direction wanted) {
        int distance = Math.abs(c.floor - floor);
        boolean onTheWay = (c.direction == Direction.UP && wanted == Direction.UP && floor >= c.floor)
                || (c.direction == Direction.DOWN && wanted == Direction.DOWN && floor <= c.floor);
        if (c.direction == Direction.IDLE || onTheWay) {
            return distance;
        }
        return distance + 100; // busy in the other direction: last resort
    }

    private final List<Car> cars = new ArrayList<>();
    private final Dispatcher dispatcher;

    public ElevatorSystem(Dispatcher dispatcher, int... startFloors) {
        this.dispatcher = dispatcher;
        for (int i = 0; i < startFloors.length; i++) {
            cars.add(new Car("E" + (i + 1), startFloors[i]));
        }
    }

    public String hallCall(int floor, Direction wanted) {
        Car car = dispatcher.choose(cars, floor, wanted);
        car.stops.add(floor);
        return "hall call " + floor + " " + wanted + " -> " + car.id;
    }

    public void carCall(String carId, int floor) {
        cars.stream().filter(c -> c.id.equals(carId)).findFirst().orElseThrow().stops.add(floor);
    }

    public List<String> tick() {
        return cars.stream().map(Car::step).toList();
    }
}

    3. Vending Machine¶

    Requirements: accept ₹1/2/5/10 coins; select a product; dispense if paid enough and in stock; return change with the fewest coins; cancel and refund. Pattern: State, with sealed types and pattern matching so each state's behaviour is explicit.

    stateDiagram-v2
    [*] --> Idle
    Idle --> HasMoney: insert coin
    HasMoney --> HasMoney: insert coin / select (not enough, sold out)
    HasMoney --> Idle: select (dispense + change)
    HasMoney --> Idle: cancel (refund)
    package com.handbook.lld.vending;

import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

/**
 * Vending machine LLD. Pattern: State. Each state decides how to react to the same events
 * (insert coin, select, cancel), which removes the big if/else on a status field.
 */
public final class VendingMachine {

    public record Product(String name, int priceRupees) {
    }

    sealed interface State permits Idle, HasMoney {
    }

    record Idle() implements State {
    }

    record HasMoney(int credit) implements State {
    }

    private static final int[] COINS = {10, 5, 2, 1};

    private final Map<String, Product> slots = new LinkedHashMap<>();
    private final Map<String, Integer> stock = new LinkedHashMap<>();
    private State state = new Idle();

    public VendingMachine load(String slot, Product product, int quantity) {
        slots.put(slot, product);
        stock.put(slot, quantity);
        return this;
    }

    public String insert(int coin) {
        if (coin != 1 && coin != 2 && coin != 5 && coin != 10) {
            return "rejected coin ₹" + coin;
        }
        state = switch (state) {
            case Idle i -> new HasMoney(coin);
            case HasMoney m -> new HasMoney(m.credit() + coin);
        };
        return "credit ₹" + ((HasMoney) state).credit();
    }

    public String select(String slot) {
        return switch (state) {
            case Idle i -> "insert coins first";
            case HasMoney m -> {
                Product p = slots.get(slot);
                if (p == null) {
                    yield "no such slot " + slot;
                }
                if (stock.get(slot) == 0) {
                    yield p.name() + " sold out, credit ₹" + m.credit() + " kept";
                }
                if (m.credit() < p.priceRupees()) {
                    yield p.name() + " costs ₹" + p.priceRupees() + ", insert ₹" + (p.priceRupees() - m.credit()) + " more";
                }
                stock.merge(slot, -1, Integer::sum);
                state = new Idle();
                yield "dispensed " + p.name() + change(m.credit() - p.priceRupees());
            }
        };
    }

    public String cancel() {
        if (state instanceof HasMoney m) {
            state = new Idle();
            return "refunded" + change(m.credit());
        }
        return "nothing to refund";
    }

    /** Greedy change-making works for canonical coin systems like 10/5/2/1. */
    private static String change(int amount) {
        if (amount == 0) {
            return "";
        }
        List<Integer> coins = new ArrayList<>();
        for (int c : COINS) {
            while (amount >= c) {
                coins.add(c);
                amount -= c;
            }
        }
        return ", change " + coins;
    }
}

    4. LRU Cache¶

    Requirements: fixed capacity; get and put in O(1); evict the least recently used entry; thread-safe. Design: HashMap<K, Node> + doubly linked list with sentinel head/tail; a lock around both.

    package com.handbook.lld.lrucache;

import java.util.HashMap;
import java.util.Map;
import java.util.StringJoiner;
import java.util.concurrent.locks.ReentrantLock;

/**
 * LRU cache with O(1) get and put: a HashMap for lookup plus a doubly linked list for recency
 * (most recent at the head). A single lock makes it thread-safe; shard it for high concurrency.
 */
public final class LruCache<K, V> {

    private static final class Node<K, V> {
        final K key;
        V value;
        Node<K, V> prev;
        Node<K, V> next;

        Node(K key, V value) {
            this.key = key;
            this.value = value;
        }
    }

    private final int capacity;
    private final Map<K, Node<K, V>> index = new HashMap<>();
    private final Node<K, V> head = new Node<>(null, null); // sentinels avoid null checks
    private final Node<K, V> tail = new Node<>(null, null);
    private final ReentrantLock lock = new ReentrantLock();
    private int evictions;

    public LruCache(int capacity) {
        if (capacity < 1) {
            throw new IllegalArgumentException("capacity must be >= 1");
        }
        this.capacity = capacity;
        head.next = tail;
        tail.prev = head;
    }

    public V get(K key) {
        lock.lock();
        try {
            Node<K, V> n = index.get(key);
            if (n == null) {
                return null;
            }
            moveToHead(n);
            return n.value;
        } finally {
            lock.unlock();
        }
    }

    public void put(K key, V value) {
        lock.lock();
        try {
            Node<K, V> n = index.get(key);
            if (n != null) {
                n.value = value;
                moveToHead(n);
                return;
            }
            if (index.size() == capacity) {
                Node<K, V> lru = tail.prev;
                unlink(lru);
                index.remove(lru.key);
                evictions++;
            }
            n = new Node<>(key, value);
            index.put(key, n);
            linkAtHead(n);
        } finally {
            lock.unlock();
        }
    }

    private void moveToHead(Node<K, V> n) {
        unlink(n);
        linkAtHead(n);
    }

    private void unlink(Node<K, V> n) {
        n.prev.next = n.next;
        n.next.prev = n.prev;
    }

    private void linkAtHead(Node<K, V> n) {
        n.next = head.next;
        n.prev = head;
        head.next.prev = n;
        head.next = n;
    }

    public int evictions() {
        return evictions;
    }

    /** Most recently used first. */
    @Override
    public String toString() {
        lock.lock();
        try {
            StringJoiner j = new StringJoiner(", ", "[", "]");
            for (Node<K, V> n = head.next; n != tail; n = n.next) {
                j.add(n.key + "=" + n.value);
            }
            return j.toString();
        } finally {
            lock.unlock();
        }
    }
}

    package com.handbook.lld.splitwise;

import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.PriorityQueue;
import java.util.TreeMap;

/**
 * Splitwise-style expense sharing. Patterns: Strategy (how an expense is split), and a greedy
 * "settle up" that minimises the number of payments by matching biggest creditor with biggest debtor.
 */
public final class ExpenseSharing {

    /** Strategy: who owes how much (in paise) of an amount. */
    public interface SplitStrategy {
        Map<String, Long> split(long amountPaise, List<String> participants);
    }

    public static SplitStrategy equal() {
        return (amount, people) -> {
            Map<String, Long> shares = new TreeMap<>();
            long base = amount / people.size();
            long remainder = amount % people.size();
            for (int i = 0; i < people.size(); i++) {
                shares.put(people.get(i), base + (i < remainder ? 1 : 0)); // spread leftover paise
            }
            return shares;
        };
    }

    public static SplitStrategy percent(Map<String, Integer> percents) {
        return (amount, people) -> {
            if (percents.values().stream().mapToInt(Integer::intValue).sum() != 100) {
                throw new IllegalArgumentException("percentages must add up to 100");
            }
            Map<String, Long> shares = new TreeMap<>();
            long assigned = 0;
            for (int i = 0; i < people.size(); i++) {
                String p = people.get(i);
                long share = i == people.size() - 1 ? amount - assigned : amount * percents.get(p) / 100;
                shares.put(p, share);
                assigned += share;
            }
            return shares;
        };
    }

    public static SplitStrategy exact(Map<String, Long> amounts) {
        return (amount, people) -> {
            if (amounts.values().stream().mapToLong(Long::longValue).sum() != amount) {
                throw new IllegalArgumentException("exact shares must add up to the total");
            }
            return new TreeMap<>(amounts);
        };
    }

    private final Map<String, Long> balance = new TreeMap<>(); // + is owed money, - owes money

    public void addExpense(String paidBy, long amountPaise, List<String> participants, SplitStrategy split) {
        balance.merge(paidBy, amountPaise, Long::sum);
        split.split(amountPaise, participants).forEach((p, share) -> balance.merge(p, -share, Long::sum));
    }

    public Map<String, Long> balances() {
        return new TreeMap<>(balance);
    }

    /** Greedy settlement: at most n-1 payments. */
    public List<String> settleUp() {
        PriorityQueue<long[]> creditors = new PriorityQueue<>((a, b) -> Long.compare(b[1], a[1]));
        PriorityQueue<long[]> debtors = new PriorityQueue<>((a, b) -> Long.compare(b[1], a[1]));
        List<String> names = new ArrayList<>(balance.keySet());
        for (int i = 0; i < names.size(); i++) {
            long b = balance.get(names.get(i));
            if (b > 0) {
                creditors.add(new long[] {i, b});
            } else if (b < 0) {
                debtors.add(new long[] {i, -b});
            }
        }
        List<String> payments = new ArrayList<>();
        while (!creditors.isEmpty() && !debtors.isEmpty()) {
            long[] c = creditors.poll();
            long[] d = debtors.poll();
            long amount = Math.min(c[1], d[1]);
            payments.add(String.format("%s pays %s ₹%,.2f", names.get((int) d[0]), names.get((int) c[0]), amount / 100.0));
            if (c[1] > amount) {
                creditors.add(new long[] {c[0], c[1] - amount});
            }
            if (d[1] > amount) {
                debtors.add(new long[] {d[0], d[1] - amount});
            }
        }
        return payments;
    }
}

    package com.handbook.lld.snakeladder;

import java.util.ArrayDeque;
import java.util.Deque;
import java.util.HashMap;
import java.util.List;

/**
 * Snake and ladder LLD. Patterns: Builder (board configuration with validation), Strategy (dice,
 * so tests can use a fixed sequence), and a queue of players for turns.
 */
public final class SnakeAndLadder {

    public static final class Board {
        final int size;
        final Map<Integer, Integer> jumps; // snake heads and ladder bottoms → destination

        private Board(int size, Map<Integer, Integer> jumps) {
            this.size = size;
            this.jumps = Map.copyOf(jumps);
        }

        public static Builder builder(int size) {
            return new Builder(size);
        }

        public static final class Builder {
            private final int size;
            private final Map<Integer, Integer> jumps = new HashMap<>();

            Builder(int size) {
                this.size = size;
            }

            public Builder snake(int head, int tail) {
                if (tail >= head) {
                    throw new IllegalArgumentException("snake must go down: " + head + "->" + tail);
                }
                return jump(head, tail);
            }

            public Builder ladder(int bottom, int top) {
                if (top <= bottom) {
                    throw new IllegalArgumentException("ladder must go up: " + bottom + "->" + top);
                }
                return jump(bottom, top);
            }

            private Builder jump(int from, int to) {
                if (from <= 1 || from >= size || to < 1 || to > size || jumps.containsKey(from)) {
                    throw new IllegalArgumentException("invalid jump " + from + "->" + to);
                }
                jumps.put(from, to);
                return this;
            }

            public Board build() {
                return new Board(size, jumps);
            }
        }
    }

    private final Board board;
    private final IntSupplier dice;
    private final Deque<String> turns;
    private final Map<String, Integer> positions = new HashMap<>();

    public SnakeAndLadder(Board board, IntSupplier dice, List<String> players) {
        this.board = board;
        this.dice = dice;
        this.turns = new ArrayDeque<>(players);
        players.forEach(p -> positions.put(p, 0));
    }

    /** Plays one turn and returns a description; ends with "WINS" when someone reaches the last square. */
    public String playTurn() {
        String player = turns.poll();
        int roll = dice.getAsInt();
        int from = positions.get(player);
        int to = from + roll;
        String note = "";
        if (to > board.size) {
            to = from;
            note = " (needs exact roll)";
        } else if (board.jumps.containsKey(to)) {
            int dest = board.jumps.get(to);
            note = dest > to ? " ladder to " + dest : " snake to " + dest;
            to = dest;
        }
        positions.put(player, to);
        turns.addLast(player);
        return String.format("%-5s rolls %d: %2d -> %2d%s%s", player, roll, from, to, note,
                to == board.size ? "  WINS" : "");
    }
}

    Requirements: books with copies; members (students: 2 loans, 14 days; faculty: 5 loans, 30 days); reservations queue when no copy is free; fines for late returns by member type; notify the next member when a reserved book comes back. Patterns: Strategy (fine policy), Observer (notifications).

import java.time.LocalDate;

/**
 * Library management LLD. Patterns: Strategy (fine policy per member type), Observer (notify the
 * next member in the reservation queue when a copy is returned).
 */
public final class Library {

    public enum MemberType { STUDENT, FACULTY }

    public record Member(String id, String name, MemberType type) {
    }

    public record Loan(String isbn, Member member, LocalDate due) {
    }

    public interface FinePolicy {
        long finePaise(Member member, long daysLate);
    }

    /** Students: ₹5/day capped at ₹200; faculty: no fines for the first 7 late days, then ₹10/day. */
    public static FinePolicy standardFines() {
        return (m, days) -> switch (m.type()) {
            case STUDENT -> Math.min(days * 500, 20_000);
            case FACULTY -> Math.max(0, days - 7) * 1_000;
        };
    }

    private final Map<String, String> titles = new HashMap<>();
    private final Map<String, Integer> available = new HashMap<>();
    private final Map<String, Queue<Member>> reservations = new HashMap<>();
    private final List<Loan> loans = new ArrayList<>();
    private final FinePolicy fines;
    private final BiConsumer<Member, String> notifier; // Observer callback (SMS/e-mail in real life)

    public Library(FinePolicy fines, BiConsumer<Member, String> notifier) {
        this.fines = fines;
        this.notifier = notifier;
    }

    public void addBook(String isbn, String title, int copies) {
        titles.put(isbn, title);
        available.merge(isbn, copies, Integer::sum);
    }

    public String borrow(Member m, String isbn, LocalDate today) {
        if (loans.stream().filter(l -> l.member().equals(m)).count() >= (m.type() == MemberType.STUDENT ? 2 : 5)) {
            return m.name() + ": loan limit reached";
        }
        Queue<Member> queue = reservations.computeIfAbsent(isbn, k -> new ArrayDeque<>());
        if (available.getOrDefault(isbn, 0) == 0) {
            if (!queue.contains(m)) {
                queue.add(m);
            }
            return m.name() + ": no copies, reservation queue position " + (List.copyOf(queue).indexOf(m) + 1);
        }
        if (!queue.isEmpty() && !queue.peek().equals(m)) {
            return m.name() + ": " + titles.get(isbn) + " is reserved for " + queue.peek().name();
        }
        queue.poll();
        available.merge(isbn, -1, Integer::sum);
        LocalDate due = today.plusDays(m.type() == MemberType.STUDENT ? 14 : 30);
        loans.add(new Loan(isbn, m, due));
        return m.name() + " borrowed " + titles.get(isbn) + ", due " + due;
    }

    public String giveBack(Member m, String isbn, LocalDate today) {
        Loan loan = loans.stream().filter(l -> l.member().equals(m) && l.isbn().equals(isbn)).findFirst().orElseThrow();
        loans.remove(loan);
        available.merge(isbn, 1, Integer::sum);
        long late = Math.max(0, ChronoUnit.DAYS.between(loan.due(), today));
        long fine = fines.finePaise(m, late);
        Queue<Member> queue = reservations.get(isbn);
        if (queue != null && !queue.isEmpty()) {
            notifier.accept(queue.peek(), titles.get(isbn) + " is ready for pickup");
        }
        return m.name() + " returned " + titles.get(isbn) + (late > 0 ? ", " + late + " days late, fine ₹" + fine / 100 : ", on time");
    }
}

    8. Movie Ticket Booking¶

    Requirements: a show's seat map; users select several seats; seats are held for 5 minutes while paying; all-or-nothing holds; confirm only while the hold is valid; expired holds become available; no double booking under concurrency. This is the core of BookMyShow-style designs.

    Free --> Held: hold (all requested seats free)
    Held --> Booked: confirm before expiry
    Held --> Free: hold expires

    package com.handbook.lld.moviebooking;

import java.time.Duration;
import java.time.Instant;
import java.util.List;
import java.util.Map;
import java.util.TreeMap;
import java.util.concurrent.ConcurrentHashMap;

/**
 * Movie ticket booking (BookMyShow-style) LLD. Core problem: two people must never get the same
 * seat. Seats are HELD for a few minutes while the user pays, then CONFIRMED or released.
 * Holds are atomic per seat (ConcurrentHashMap.compute); expired holds are reclaimable.
 */
public final class ShowBooking {

    public enum Status { HELD, BOOKED }

    public record SeatState(Status status, String user, Instant holdExpires) {
    }

    private final Map<String, SeatState> seats = new ConcurrentHashMap<>();
    private final List<String> layout;
    private final Duration holdTime;

    public ShowBooking(List<String> layout, Duration holdTime) {
        this.layout = List.copyOf(layout);
        this.holdTime = holdTime;
    }

    /** Hold all requested seats or none (seats are always processed in sorted order to avoid deadlock-like partial holds). */
    public synchronized boolean hold(String user, List<String> wanted, Instant now) {
        for (String seat : wanted) {
            if (!layout.contains(seat) || !isFree(seat, now)) {
                return false;
            }
        }
        wanted.stream().sorted().forEach(seat -> seats.put(seat, new SeatState(Status.HELD, user, now.plus(holdTime))));
        return true;
    }

    public synchronized boolean confirm(String user, List<String> held, Instant now) {
        for (String seat : held) {
            SeatState s = seats.get(seat);
            if (s == null || s.status() != Status.HELD || !s.user().equals(user) || !now.isBefore(s.holdExpires())) {
                return false; // hold lost (expired or never ours): payment must be refunded
            }
        }
        held.forEach(seat -> seats.put(seat, new SeatState(Status.BOOKED, user, null)));
        return true;
    }

    private boolean isFree(String seat, Instant now) {
        SeatState s = seats.get(seat);
        return s == null || (s.status() == Status.HELD && !now.isBefore(s.holdExpires()));
    }

    public synchronized String seatMap(Instant now) {
        Map<String, String> view = new TreeMap<>();
        for (String seat : layout) {
            SeatState s = seats.get(seat);

import java.util.ArrayList;
import java.util.EnumMap;
import java.util.List;
import java.util.Map;
import java.util.Set;

/**
 * Notification system LLD. Patterns: Strategy (one sender per channel), Factory/registry (channel →
 * sender), Template Method (every message renders the same way with channel-specific formatting),
 * Decorator (retry around any sender), and user preferences as data.
 */
public final class NotificationService {

    public enum Channel { EMAIL, SMS, PUSH }

    public record User(String id, String name, String email, String phone, Set<Channel> optedIn) {
    }

    /** Strategy. */
    public interface Sender {
        void send(User user, String text);
    }

    /** Template Method: fixed steps, channel-specific formatting hook. */
    public abstract static class MessageTemplate {
        public final String render(User user, Map<String, String> data) {
            return format(greeting(user) + body(data)); // the skeleton never changes
        }

        protected String greeting(User user) {
            return "Hi " + user.name() + ", ";
        }

        protected abstract String body(Map<String, String> data);

        protected String format(String text) {
            return text;
        }
    }

    /** Decorator: retries any sender a few times on failure. */
    public static Sender withRetry(Sender inner, int attempts, List<String> log) {
        return (user, text) -> {
            for (int i = 1; ; i++) {
                try {
                    inner.send(user, text);
                    return;
                } catch (RuntimeException e) {
                    log.add("retry " + i + " after: " + e.getMessage());
                    if (i == attempts) {
                        throw e;
                    }
                }
            }
        };
    }

    private final Map<Channel, Sender> senders = new EnumMap<>(Channel.class);
    private final List<String> failures = new ArrayList<>();

    public NotificationService register(Channel channel, Sender sender) {
        senders.put(channel, sender);
        return this;
    }

    public void notify(User user, MessageTemplate template, Map<String, String> data) {
        String text = template.render(user, data);
        for (Channel c : Channel.values()) {
            if (!user.optedIn().contains(c)) {
                continue; // respect preferences
            }
            try {
                senders.get(c).send(user, c == Channel.SMS && text.length() > 60 ? text.substring(0, 57) + "..." : text);
            } catch (RuntimeException e) {
                failures.add(c + " to " + user.id() + " failed: " + e.getMessage()); // one channel failing doesn't stop others
            }
        }
    }

    public List<String> failures() {
        return List.copyOf(failures);
    }
}

    package com.handbook.lld.logging;

import java.time.Clock;
import java.util.Map;
import java.util.TreeMap;
import java.util.function.Consumer;

/**
 * Logging framework LLD. Patterns: Chain of Responsibility (each appender handles the record if its
 * level allows, then passes it on), Strategy (formatter), and a hierarchy of named loggers
 * ("com.shop.payment" inherits settings from "com.shop" unless overridden).
 */
public final class MiniLogger {

    public enum Level { DEBUG, INFO, WARN, ERROR }

    public record LogRecord(long epochMillis, Level level, String logger, String message) {
    }

    /** Strategy for turning a record into text. */
    public interface Formatter {
        String format(LogRecord r);
    }

    /** A link in the chain. */
    public static final class Appender {
        private final String name;
        private final Level threshold;
        private final Formatter formatter;
        private final Consumer<String> sink;
        private Appender next;

        public Appender(String name, Level threshold, Formatter formatter, Consumer<String> sink) {
            this.name = name;
            this.threshold = threshold;
            this.formatter = formatter;
            this.sink = sink;
        }

        public Appender then(Appender next) {
            this.next = next;
            return next;
        }

        void handle(LogRecord r) {
            if (r.level().compareTo(threshold) >= 0) {
                sink.accept("[" + name + "] " + formatter.format(r));
            }
            if (next != null) {
                next.handle(r);
            }
        }
    }

    private final Clock clock;
    private final Appender chain;
    private final Map<String, Level> levels = new TreeMap<>();

    public MiniLogger(Clock clock, Appender chain, Level rootLevel) {
        this.clock = clock;
        this.chain = chain;
        levels.put("", rootLevel);
    }

    public void setLevel(String loggerName, Level level) {
        levels.put(loggerName, level);
    }

    /** The most specific configured prefix wins: com.shop.payment → com.shop → root. */
    Level effectiveLevel(String loggerName) {
        String name = loggerName;
        while (true) {
            Level l = levels.get(name);
            if (l != null) {
                return l;
            }
            int dot = name.lastIndexOf('.');
            name = dot < 0 ? "" : name.substring(0, dot);
        }
    }

    public void log(String loggerName, Level level, String message) {
        if (level.compareTo(effectiveLevel(loggerName)) < 0) {
            return; // cheap early exit: no record, no formatting
        }
        chain.handle(new LogRecord(clock.millis(), level, loggerName, message));
    }
}

    11. Grocery Discount Engine¶

    Requirements: rules like "buy 2 get 1 free" on a SKU, "10% off dairy for Gold members", "₹150 off over ₹1,000", "5% off over ₹2,000"; some rules stack, while among exclusive coupons only the best applies; the total never goes negative; new rules without changing the engine. Patterns: Specification (eligibility), Strategy (discount calculation).

    package com.handbook.lld.discount;

import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Map;
import java.util.function.Predicate;
import java.util.function.ToLongFunction;

/**
 * Grocery discount engine LLD. Patterns: Specification (eligibility predicates that combine with
 * and/or), Strategy (how a rule computes its discount), and a policy for combining rules:
 * stackable rules all apply; among exclusive rules only the best one does.
 */
public final class DiscountEngine {

    public record Item(String sku, String category, int qty, long unitPaise) {
        long total() {
            return qty * unitPaise;
        }
    }

    public record Cart(String customerTier, List<Item> items) {
        long subtotal() {
            return items.stream().mapToLong(Item::total).sum();
        }
    }

    /** Specification: is the rule applicable to this cart? */
    public interface Spec extends Predicate<Cart> {
        default Spec and(Spec other) {
            return c -> test(c) && other.test(c);
        }

        static Spec subtotalAtLeast(long paise) {
            return c -> c.subtotal() >= paise;
        }

        static Spec tier(String tier) {
            return c -> c.customerTier().equals(tier);
        }

        static Spec contains(String sku) {
            return c -> c.items().stream().anyMatch(i -> i.sku().equals(sku));
        }
    }

    public record Rule(String name, Spec when, ToLongFunction<Cart> discount, boolean exclusive) {
    }

    public record Applied(String rule, long paise) {
    }

    /** Strategies for common rules. */
    public static ToLongFunction<Cart> percentOffCategory(String category, int percent) {
        return c -> c.items().stream().filter(i -> i.category().equals(category)).mapToLong(Item::total).sum() * percent / 100;
    }

    public static ToLongFunction<Cart> buyXGetYFree(String sku, int x, int y) {
        return c -> c.items().stream().filter(i -> i.sku().equals(sku))
                .mapToLong(i -> (long) (i.qty() / (x + y)) * y * i.unitPaise()).sum();
    }

    public static ToLongFunction<Cart> flat(long paise) {
        return c -> paise;
    }

    private final List<Rule> rules = new ArrayList<>();

    public DiscountEngine add(Rule rule) {
        rules.add(rule);
        return this;
    }

    public List<Applied> apply(Cart cart) {
        List<Applied> applied = new ArrayList<>();
        for (Rule r : rules) {
            if (!r.exclusive() && r.when().test(cart)) {
                long d = r.discount().applyAsLong(cart);
                if (d > 0) {
                    applied.add(new Applied(r.name(), d));
                }
            }
        }
        rules.stream().filter(Rule::exclusive).filter(r -> r.when().test(cart))
                .map(r -> new Applied(r.name(), r.discount().applyAsLong(cart)))
                .filter(a -> a.paise() > 0)
                .max(Comparator.comparingLong(Applied::paise))
                .ifPresent(applied::add); // best exclusive offer only
        return applied;
    }

    public static Map<String, Long> summary(Cart cart, List<Applied> applied) {
        long discount = Math.min(cart.subtotal(), applied.stream().mapToLong(Applied::paise).sum()); // never negative totals
        return Map.of("subtotal", cart.subtotal(), "discount", discount, "total", cart.subtotal() - discount);
    }
}

    package com.handbook.lld.tictactoe;

/**
 * Tic-tac-toe on an n×n board with O(1) win detection: keep a running sum per row, column and
 * diagonal (+1 for X, -1 for O). A line wins when its sum reaches ±n.
 */
public final class TicTacToe {

    public enum Result { IN_PROGRESS, X_WINS, O_WINS, DRAW }

    private final int n;
    private final char[][] cells;
    private final int[] rows;
    private final int[] cols;
    private int diagonal;
    private int antiDiagonal;
    private int moves;
    private char turn = 'X';
    private Result result = Result.IN_PROGRESS;

    public TicTacToe(int n) {
        this.n = n;
        this.cells = new char[n][n];
        this.rows = new int[n];
        this.cols = new int[n];
    }

    public Result move(int r, int c) {
        if (result != Result.IN_PROGRESS) {
            throw new IllegalStateException("game over: " + result);
        }
        if (r < 0 || c < 0 || r >= n || c >= n || cells[r][c] != 0) {
            throw new IllegalArgumentException("illegal move " + r + "," + c);
        }
        cells[r][c] = turn;
        int delta = turn == 'X' ? 1 : -1;
        rows[r] += delta;
        cols[c] += delta;
        if (r == c) {
            diagonal += delta;
        }
        if (r + c == n - 1) {
            antiDiagonal += delta;
        }
        moves++;
        if (Math.abs(rows[r]) == n || Math.abs(cols[c]) == n || Math.abs(diagonal) == n || Math.abs(antiDiagonal) == n) {
            result = turn == 'X' ? Result.X_WINS : Result.O_WINS;
        } else if (moves == n * n) {
            result = Result.DRAW;
        }
        turn = turn == 'X' ? 'O' : 'X';
        return result;
    }

    public String board() {
        StringBuilder sb = new StringBuilder();
        for (char[] row : cells) {
            for (char cell : row) {
                sb.append(cell == 0 ? '.' : cell);
            }
            sb.append('\n');
        }
        return sb.toString();
    }
}

import com.handbook.lld.vending.VendingMachine;
import org.junit.jupiter.api.Test;

import java.time.Duration;
import java.time.Instant;
import java.util.List;
import java.util.Map;

import static org.assertj.core.api.Assertions.assertThat;
import static org.assertj.core.api.Assertions.assertThatThrownBy;

class LldProblemsTest {

    private static final Instant T0 = Instant.parse("2026-01-01T00:00:00Z");

    @Test
    void parkingLotNeverPutsACarInASmallSpot() {
        var lot = new ParkingLot(ParkingLot.nearestSmallestFit(), (t, d) -> 0).addFloor(0, 2, 0, 0);
        assertThat(lot.park(new Vehicle("C", VehicleType.CAR), T0)).isEmpty();
        assertThat(lot.park(new Vehicle("B", VehicleType.BIKE), T0)).isPresent();
    }

    @Test
    void vendingMachineGivesCorrectChange() {
        var vm = new VendingMachine().load("A", new VendingMachine.Product("x", 13), 1);
        vm.insert(10);
        vm.insert(10);
        assertThat(vm.select("A")).endsWith("change [5, 2]");
    }

    @Test
    void lruEvictsLeastRecentlyUsed() {
        var c = new LruCache<String, Integer>(2);
        c.put("a", 1);
        c.put("b", 2);
        c.get("a");
        c.put("c", 3);
        assertThat(c.get("b")).isNull();
        assertThat(c.get("a")).isEqualTo(1);
    }

    @Test
    void expenseBalancesAlwaysSumToZero() {
        var s = new ExpenseSharing();
        s.addExpense("a", 1_001, List.of("a", "b", "c"), ExpenseSharing.equal());
        assertThat(s.balances().values().stream().mapToLong(Long::longValue).sum()).isZero();
        assertThat(s.settleUp()).hasSize(2);
    }

    @Test
    void expiredHoldCanBeTakenByAnotherUser() {
        var show = new ShowBooking(List.of("A1"), Duration.ofMinutes(5));
        assertThat(show.hold("u1", List.of("A1"), T0)).isTrue();
        assertThat(show.hold("u2", List.of("A1"), T0.plusSeconds(60))).isFalse();
        assertThat(show.hold("u2", List.of("A1"), T0.plusSeconds(301))).isTrue();
        assertThat(show.confirm("u1", List.of("A1"), T0.plusSeconds(302))).isFalse();
    }

    @Test
    void onlyBestExclusiveDiscountApplies() {
        var engine = new DiscountEngine()
                .add(new Rule("a", Spec.subtotalAtLeast(0), DiscountEngine.flat(100), true))
                .add(new Rule("b", Spec.subtotalAtLeast(0), DiscountEngine.flat(300), true));
        var applied = engine.apply(new Cart("R", List.of(new Item("x", "c", 1, 1_000))));
        assertThat(applied).extracting(DiscountEngine.Applied::rule).containsExactly("b");
    }

    @Test
    void ticTacToeDetectsColumnWin() {
        var g = new TicTacToe(3);
        g.move(0, 0);
        g.move(0, 1);
        g.move(1, 0);
        g.move(1, 1);
        assertThat(g.move(2, 0)).isEqualTo(TicTacToe.Result.X_WINS);
        assertThatThrownBy(() -> g.move(2, 2)).isInstanceOf(IllegalStateException.class);
    }

    @Test
    void consistentHashingIsStable() {
        var ring = new ConsistentHashRing(50);
        ring.addNode("n1");
        ring.addNode("n2");
        String before = ring.nodeFor("k");
        assertThat(ring.nodeFor("k")).isEqualTo(before);
        assertThat(Map.of("n1", 1, "n2", 1)).containsKey(before);
    }
}
